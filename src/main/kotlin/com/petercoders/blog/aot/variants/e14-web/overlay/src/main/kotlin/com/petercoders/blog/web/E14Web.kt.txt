// e14 변형 앱 전용. variants/build-variant.sh 가 f3c9bfb 트리에 .kt 로 복사해 빌드한다.
// (메인 빌드에 섞이지 않도록 저장소에서는 .kt.txt 로 둔다)
//
// JSON 직렬화(GET 둘)와 역직렬화(POST 하나)가 있는 엔드포인트 셋, 그리고 HotSpotAOTCacheMXBean.endRecording()
// 으로 training 을 끝내는 엔드포인트 하나(e14 b'' 변형에서만 부른다).
package com.petercoders.blog.web

import jdk.management.HotSpotAOTCacheMXBean
import org.springframework.web.bind.annotation.GetMapping
import org.springframework.web.bind.annotation.PathVariable
import org.springframework.web.bind.annotation.PostMapping
import org.springframework.web.bind.annotation.RequestBody
import org.springframework.web.bind.annotation.RequestMapping
import org.springframework.web.bind.annotation.RequestParam
import org.springframework.web.bind.annotation.RestController
import java.lang.management.ManagementFactory

data class Item(val id: Long, val name: String, val tags: List<String>, val price: Double)
data class Order(val id: Long, val items: List<Item>, val total: Double)

@RestController
@RequestMapping("/api")
class E14Controller {

    @GetMapping("/items/{id}")
    fun item(@PathVariable id: Long): Item = Item(id, "item-$id", listOf("a", "b"), id * 1.5)

    @GetMapping("/orders")
    fun orders(@RequestParam(defaultValue = "3") n: Int): List<Order> =
        (1..n).map { o -> Order(o.toLong(), (1..3).map { Item(it.toLong(), "i$it", listOf("x"), it * 2.0) }, 6.0 * o) }

    @PostMapping("/echo")
    fun echo(@RequestBody body: Map<String, Any?>): Map<String, Any?> = body + ("size" to body.size)

    @PostMapping("/aot/end")
    fun endRecording(): Map<String, Any> {
        val mx = ManagementFactory.getPlatformMXBean(HotSpotAOTCacheMXBean::class.java)
        val wasRecording = mx.isRecording
        return mapOf("mode" to mx.mode, "wasRecording" to wasRecording, "ended" to mx.endRecording())
    }
}
